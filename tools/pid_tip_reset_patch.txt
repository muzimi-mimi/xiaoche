/* ==========================================================================
 * PID 状态复位 / 倾角重对准
 *
 * PID_ResetAll()：把三个环的 Target / Out / 积分 / 误差全部清零，并把"上次值"
 *   重新播种为当前传感器读数，避免下一轮微分项第一次冲击。
 *   倾倒(StopCar)、急停 "0"、起立超时都会经过 StopCar，顺带清干净。
 *   函数体内只有浮点赋值和全局变量读取，可以从 1ms 中断里安全快速调用。
 *
 * Angle_Realign()：用加速度计把倾角重新对准一次(静止时可信)。
 *   互补滤波 Alpha=0.001，时间常数约 10s：摔倒时陀螺饱和会把 Angle 积到
 *   偏离真实值很远，若不重对准，按 "1" 后可能因为 |Angle| 误报成很小而直接
 *   跳进 BALANCE —— 车还躺着，程序却以为已经站起来了。
 *   内含 30 x 10ms = 300ms 的 HAL_Delay，只能在主循环调用，
 *   绝对不能放进 1ms 的 HAL_TIM_PeriodElapsedCallback。
 * ========================================================================== */
void PID_ResetAll(void)
{
    PID_Init(&AnglePID);
    PID_Init(&SpeedPID);
    PID_Init(&TurnPID);

    /* 预置"上次值"，防止微分项第一次冲击 */
    AnglePID.Actual  = Angle;
    AnglePID.Actual1 = Angle;
    AnglePID.Error0  = AnglePID.Target - AnglePID.Actual;
    AnglePID.Error1  = AnglePID.Error0;

    SpeedPID.Actual  = AveSpeed;
    SpeedPID.Actual1 = AveSpeed;
    TurnPID.Actual   = DifSpeed;
    TurnPID.Actual1  = DifSpeed;

    DifSpeedFilter = DifSpeed;
    DifPWM         = 0;
}

void Angle_Realign(void)
{
    int32_t  sum = 0;
    int      k;
    uint32_t itWasOn;

    /* 采样期间屏蔽 TIM3 中断：中断里也会读 MPU6050(同一条 I2C)，
     * 并发访问会互相破坏采样。只动 DIER 的 UIE 位，启动状态保持原样，
     * 所以上电初始化和运行中调用都适用。 */
    itWasOn = (htim3.Instance->DIER & TIM_IT_UPDATE) ? 1u : 0u;
    __HAL_TIM_DISABLE_IT(&htim3, TIM_IT_UPDATE);

    for (k = 0; k < 30; k++)              /* 30 x 10ms = 300ms */
    {
        MPU6050_GetData(&AX, &AY, &AZ, &GX, &GY, &GZ);
        sum += (int32_t)(-atan2(AY, AZ) / 3.14159 * 180.0);
        HAL_Delay(10);
    }

    Angle = (float)sum / 30.0f + G;       /* 取平均，抑制振动噪声 */

    if (itWasOn)
    {
        __HAL_TIM_CLEAR_IT(&htim3, TIM_IT_UPDATE);
        __HAL_TIM_ENABLE_IT(&htim3, TIM_IT_UPDATE);
    }
}

/* ==========================================================================
 * 状态机动作：StartErect() / StopCar()
 * StartErect() 只在主循环（蓝牙解析处）调用；StopCar() 主循环和 10ms 中断
 * 都会调用，它只做赋值、PID 复位和 GPIO 写，可以从中断里安全调用。
 * CarState 一律最后写：中断只会看到"旧状态"或"新状态已就绪"两种，
 * 所以不需要额外的临界区。
 * ========================================================================== */
void StopCar(void)
{
    CarState  = ST_IDLE;
    ErectTime = 0;
    PID_ResetAll();               /* 倾倒/急停/超时：三个环的内部状态一起清干净 */
    TB6612_setPWM(1, 0);          /* 停：PWM=0，IN1/IN2 同向 -> 电机滑行 */
    TB6612_setPWM(2, 0);
}

void StartErect(void)
{
    if (CarState != ST_IDLE)
    {
        return;                   /* 起立中或已直立：忽略重复的 "1" */
    }

    /* 先重对准倾角，再交给自起立：躺倒后按 "1" 拿到的是真实倾角，
     * 而不是漂掉的陀螺积分值。300ms 阻塞，只在主循环里跑。 */
    Angle_Realign();

    PID_ResetAll();
    ErectTime = 0;

    CarState = ST_ERECT;          /* 最后一步：从中断视角看是原子的 */
}
